# ============================================================
# NOVA backend — Phase 1 dependencies
# Each dependency is here for a stated reason (see README).
# ============================================================

# Web framework: async HTTP API, dependency injection, OpenAPI docs at /docs
fastapi>=0.115,<1.0
# ASGI server that runs the FastAPI app (standard extras add --reload, websockets)
uvicorn[standard]>=0.30,<1.0
# Data validation / settings; FastAPI's schema layer is built on it
pydantic>=2.9,<3.0
# Typed .env loading for pydantic settings (pulls python-dotenv transitively)
pydantic-settings>=2.6,<3.0

# NLP toolkit: tokenizers, stopwords, Porter stemmer, WordNet lemmatizer
# (corpora are fetched separately via `python -m app.nlp.nltk_data`)
nltk>=3.9,<4.0

# --- Phase 3: intent classification ---
# Dense arrays under every model/TF-IDF operation (direct import in nlp code)
numpy>=1.26,<3.0
# TF-IDF vectorization + LogisticRegression / Naive Bayes / LinearSVM classifiers
scikit-learn>=1.5,<2.0
# Persisting/loading the trained vectorizer + model (direct import, not via sklearn)
joblib>=1.4,<2.0
# Notebook data display + evaluation report tables
pandas>=2.2,<4.0

# --- Phase 4: entity extraction ---
# Statistical NER (pipeline: tokenizer/POS/NER); the model is downloaded
# separately via `python -m spacy download en_core_web_sm` — without it the
# entity extractor degrades to the built-in rule-based fallback
spacy>=3.8,<4.0

# --- Phase 5: sentence embeddings (semantic search / RAG) ---
# all-MiniLM-L6-v2 (~80 MB) downloads to the HF cache on first encode;
# if unavailable, callers degrade to the TF-IDF chain
sentence-transformers>=3.0,<7.0

# --- Phase 6: conversation memory ---
# ORM + engine for Conversation/Message tables; the store falls back to
# in-memory when NOVA_DATABASE_URL is unset or unreachable
sqlalchemy>=2.0,<3.0
# PostgreSQL driver for SQLAlchemy (`postgresql+psycopg://` URLs);
# psycopg is also used directly by Phase 7's pgvector queries
psycopg[binary]>=3.2,<4.0

# --- Test-only ---
pytest>=8.0
# httpx2 is the maintained successor of httpx; starlette's TestClient
# (fastapi.testclient) uses it and warns on plain httpx
httpx2>=2.10

# --- Phase 7: retrieval-augmented generation ---
# PDF text extraction (document upload pipeline); txt/md use stdlib decode
pypdf>=5.0,<7.0
# multipart form parsing for FastAPI File uploads (documents endpoint)
python-multipart>=0.0.9,<1.0

# --- Phase 8: LLM providers ---
# HTTP client for provider APIs (OpenAI/HuggingFace); injectable transport
# lets every provider test run offline via httpx.MockTransport
httpx>=0.27,<1.0
# LocalModelProvider loads a transformers text-generation pipeline lazily
# (already installed via sentence-transformers; declared as a direct import)
transformers>=4.40,<6.0
